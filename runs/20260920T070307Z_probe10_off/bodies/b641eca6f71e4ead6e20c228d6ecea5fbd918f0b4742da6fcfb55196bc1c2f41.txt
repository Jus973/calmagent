    def get_possible_moves(self, state):
        blank = self.find_blank(state)
        if not blank:
            return None

        i, j = blank
        possible_moves = []

        if i > 0:
            possible_moves.append('up')
        if i < len(state) - 1:
            possible_moves.append('down')
        if j > 0:
            possible_moves.append('left')
        if j < len(state[0]) - 1:
            possible_moves.append('right')

        return possible_moves
