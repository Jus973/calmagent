    def vigenere_cipher(self, text):
        if not text:
            return ""

        key = self.key
        key_length = len(key)
        result = []
        key_index = 0

        for char in text:
            if char.isalpha():
                shift_amount = ord(key[key_index].lower()) - ord('a')
                base = ord('a') if char.islower() else ord('A')
                shifted_char = chr((ord(char) - base + shift_amount) % 26 + base)
                result.append(shifted_char)
                key_index = (key_index + 1) % key_length
            else:
                result.append(char)

        return ''.join(result)
