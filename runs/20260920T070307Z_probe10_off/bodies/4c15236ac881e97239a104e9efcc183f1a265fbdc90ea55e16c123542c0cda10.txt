    def rail_fence_cipher(self, text, num_rails):
        if not text:
            return ""

        if num_rails == 1:
            return text

        rails = [''] * num_rails
        rail_index = 0
        direction = 1

        for char in text:
            rails[rail_index] += char
            if rail_index == 0:
                direction = 1
            elif rail_index == num_rails - 1:
                direction = -1
            rail_index += direction

        return ''.join(rails)
